cmake_minimum_required(VERSION 3.16)
project(seedbox_ic_fixtures C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
find_package(DynamoRIO REQUIRED)
add_library(ic_observer SHARED ic-observer.c)
configure_DynamoRIO_client(ic_observer)
use_DynamoRIO_extension(ic_observer drmgr)
add_executable(ic-dependencies ic-dependencies.c)

if(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64")
  enable_language(ASM)
  add_executable(parked-raw-context parked-raw-context.c)
  target_link_libraries(parked-raw-context pthread)
  add_library(parked_raw_client SHARED parked-raw-client.c)
  configure_DynamoRIO_client(parked_raw_client)
  add_executable(parked-publication-context parked-publication-context.c
                 parked-publication-context.S)
  target_link_libraries(parked-publication-context pthread)
  add_library(parked_publication_client SHARED parked-publication-client.c)
  configure_DynamoRIO_client(parked_publication_client)
endif()
