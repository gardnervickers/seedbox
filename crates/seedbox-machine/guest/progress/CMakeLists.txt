cmake_minimum_required(VERSION 3.16)
project(seedbox_progress C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(DynamoRIO_USE_LIBC OFF)
find_package(DynamoRIO REQUIRED)
add_library(seedbox_progress SHARED progress.c)
configure_DynamoRIO_client(seedbox_progress)
use_DynamoRIO_extension(seedbox_progress drmgr)
use_DynamoRIO_extension(seedbox_progress drreg)
