foreach(case IN LISTS unix_shared)
    add_executable(unix_${case} ${case}.c)
    set_target_properties(unix_${case} PROPERTIES OUTPUT_NAME unix-${case})
    target_compile_definitions(unix_${case} PRIVATE UNIX_SH_PATH="${UNIX_SH_PATH}" UNIX_KILL_PATH="${UNIX_KILL_PATH}")
    target_compile_options(unix_${case} PRIVATE -O2 -g -fno-omit-frame-pointer)
    target_link_options(unix_${case} PRIVATE -rdynamic)
    target_link_libraries(unix_${case} PRIVATE pthread dl rt)
    install(TARGETS unix_${case} RUNTIME DESTINATION bin)
endforeach()
