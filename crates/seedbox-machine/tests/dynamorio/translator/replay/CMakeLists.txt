# The parent has already found the matching public DynamoRIO SDK.
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_executable(translator_replay_shared shared.c)
configure_DynamoRIO_decoder(translator_replay_shared)
install(TARGETS translator_replay_shared RUNTIME DESTINATION bin)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64")
  foreach(case IN ITEMS decode encode semantics ir_transform selectors)
    add_executable(translator_replay_${case} ${case}.c)
    configure_DynamoRIO_decoder(translator_replay_${case})
    install(TARGETS translator_replay_${case} RUNTIME DESTINATION bin)
  endforeach()
endif()
# Also exercise the actual standalone core, including its Debug assertions.
# The SDK's drdecode target intentionally disables some core assertions, even
# in a Debug SDK. No private assertion hooks or release fallback are used.
set(replay_standalone_cases shared)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64")
  list(APPEND replay_standalone_cases decode encode semantics ir_transform selectors)
endif()
set(DynamoRIO_RPATH OFF)
foreach(case IN LISTS replay_standalone_cases)
  add_executable(translator_replay_${case}_standalone ${case}.c)
  configure_DynamoRIO_standalone(translator_replay_${case}_standalone)
  set_property(TARGET translator_replay_${case}_standalone PROPERTY BUILD_RPATH "$<TARGET_FILE_DIR:dynamorio>")
  set_property(TARGET translator_replay_${case}_standalone PROPERTY INSTALL_RPATH "$<TARGET_FILE_DIR:dynamorio>")
  install(TARGETS translator_replay_${case}_standalone RUNTIME DESTINATION bin)
endforeach()
install(FILES cases.tsv source_mapping.tsv provenance.txt hashes.json DynamoRIO-LICENSE.txt DESTINATION share/translator/replay)

  configure_DynamoRIO_decoder(translator_replay_feature_queries)
  install(TARGETS translator_replay_feature_queries RUNTIME DESTINATION bin)
endif()
